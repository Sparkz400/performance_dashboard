===== ..\src\App.jsx =====
import React, { useEffect, useState } from 'react';
import {
    BrowserRouter as Router,
    Routes,
    Route
} from 'react-router-dom';
import {
    Box,
    CssBaseline,
    AppBar,
    Toolbar,
    Typography,
    createTheme,
    ThemeProvider
} from '@mui/material';
import io from 'socket.io-client';

import ResponsiveSidebar from './components/sidebar/sidebar.jsx';
import DashboardManager from './pages/dashboardPage.jsx';
import SettingsPage from './pages/settingsPage';
import HelpPage from './pages/helpPage';

const darkTheme = createTheme({
    palette: {
        mode: 'dark',
        primary: {
            main: '#2b2d30'
        },
        background: {
            default: '#1e1f22',
            paper: '#1e1f22'
        }
    }
});

function App() {
    const [sensorData, setSensorData] = useState([]);
    const [status, setStatus] = useState({ database: 'unknown', opcua: 'unknown' });
    const [sidebarOpen, setSidebarOpen] = useState(true);
    const [drawerWidth, setDrawerWidth] = useState(0);

    useEffect(() => {
        const socket = io('http://localhost:3000');
        socket.on('sensorUpdate', (data) => {
            setSensorData((prev) => [data, ...prev]);
        });
        return () => socket.disconnect();
    }, []);

    useEffect(() => {
        async function fetchStatus() {
            try {
                const res = await fetch('http://localhost:3000/api/status');
                const data = await res.json();
                setStatus(data);
            } catch (err) {
                console.error('Error fetching status:', err);
            }
        }
        fetchStatus();
        const interval = setInterval(fetchStatus, 5000);
        return () => clearInterval(interval);
    }, []);

    const toggleSidebar = () => {
        setSidebarOpen((prev) => !prev);
    };

    return (
        <ThemeProvider theme={darkTheme}>
            <Router>
                <Box sx={{ display: 'flex', minHeight: '100vh' }}>
                    <CssBaseline />
                    <ResponsiveSidebar
                        open={sidebarOpen}
                        onToggle={toggleSidebar}
                        onWidthChange={(width) => setDrawerWidth(width)}
                    />
                    <Box sx={{
                        flexGrow: 1,
                        ml: `${drawerWidth}px`,
                        overflow: 'auto',
                        maxHeight: '100vh'
                    }}>
                        <Routes>
                            <Route path="/" element={<DashboardManager status={status} sensorData={sensorData} />} />
                            <Route path="/settings" element={<SettingsPage />} />
                            <Route path="/help" element={<HelpPage />} />
                        </Routes>
                    </Box>
                </Box>
            </Router>
        </ThemeProvider>
    );
}

export default App;

import fs from 'fs';
import path from 'path';
import { fileURLToPath } from 'url';

// Convert the module URL to a file path
const __filename = fileURLToPath(import.meta.url);
const __dirname = path.dirname(__filename);

const extensions = [".jsx", ".css", ".js"];
const excludedDirs = new Set(["node_modules", "public", ".idea"]);
const outputFile = "all_code.txt";

function walkDir(dir, callback) {
    fs.readdirSync(dir).forEach(file => {
        const dirPath = path.join(dir, file);
        const isDirectory = fs.statSync(dirPath).isDirectory();
        if (isDirectory && !excludedDirs.has(file)) {
            walkDir(dirPath, callback);
        } else if (extensions.some(ext => file.endsWith(ext))) {
            callback(dirPath);
        }
    });
}

const outputStream = fs.createWriteStream(outputFile, { encoding: 'utf-8' });

walkDir('..', (filePath) => {
    const content = fs.readFileSync(filePath, { encoding: 'utf-8', flag: 'r' });
    outputStream.write(`===== ${filePath} =====\n`);
    outputStream.write(content + '\n\n');
});

outputStream.end();
console.log(`Code copied to ${outputFile}, excluding ${[...excludedDirs].join(', ')}`);

import React, { useCallback, useEffect, useRef } from 'react';
import { Link } from 'react-router-dom';
import {
    Box,
    Divider,
    Drawer,
    List,
    ListItem, ListItemButton,
    ListItemIcon,
    ListItemText,
    Toolbar
} from '@mui/material';
import { useTheme } from '@mui/material/styles';

// Icons
import DashboardIcon from '@mui/icons-material/Dashboard';
import SettingsIcon from '@mui/icons-material/Settings';
import HelpOutlineIcon from '@mui/icons-material/HelpOutline';

import BurgerButton from './burgerButton.jsx';

const ResponsiveSidebar = ({ open, onToggle, onWidthChange }) => {
    const theme = useTheme();
    const contentRef = useRef(null);

    // measureWidth calculates how wide the content is (text + icons, if open),
    // or icons only if open=false. We clamp to 50% viewport as an example.
    const measureWidth = useCallback(() => {
        if (!contentRef.current) return 0;
        const contentWidth = contentRef.current.scrollWidth;
        const maxSidebarWidth = window.innerWidth * 0.5;
        return Math.min(contentWidth, maxSidebarWidth);
    }, []);

    useEffect(() => {
        // Re-measure whenever open changes
        if (open) {
            // Let the text be visible, then measure
            requestAnimationFrame(() => {
                const w = measureWidth();
                onWidthChange(w);
            });
        } else {
            // With text hidden, measure again
            requestAnimationFrame(() => {
                const w = measureWidth();
                onWidthChange(w);
            });
        }

        const handleResize = () => {
            const w = measureWidth();
            onWidthChange(w);
        };
        window.addEventListener('resize', handleResize);
        return () => window.removeEventListener('resize', handleResize);
    }, [open, measureWidth, onWidthChange]);

    return (
        <Drawer
            variant="persistent"
            open={true}  // The drawer is always "open" from MUI's perspective, but we control the width with measure
            PaperProps={{
                sx: {
                    position: 'fixed',
                    overflowX: 'hidden',
                    height: '100vh',
                }
            }}
            sx={{ flexShrink: 0 }}
        >
            <Divider />

            <Box
                ref={contentRef}
                sx={{
                    display: 'flex',          // flex layout
                    flexDirection: 'column',  // stack items vertically
                    height: '100%',           // full available height
                    backgroundColor: theme.palette.primary.main,
                    color: theme.palette.common.white,
                    whiteSpace: 'nowrap',
                    py: 1,
                    px: 0
                }}
            >
                <List disablePadding
                      sx={{
                          // Let the List fill all vertical space
                          flexGrow: 1,
                          display: 'flex',
                          flexDirection: 'column'
                      }}
                >
                    {/* 1) Burger Button (at the top) */}
                    <ListItem disablePadding
                              sx={{
                                  display: 'flex',
                                  alignItems: 'center',
                                  px: 2,
                                  py: 1
                              }}
                    >
                        <BurgerButton checked={open} onChange={onToggle} />
                    </ListItem>
                    <Divider />

                    {/* B.2) Main (top) navigation items */}
                    <ListItem disablePadding>
                        <ListItemButton component={Link} to="/">
                            <ListItemIcon sx={{ color: theme.palette.common.white }}>
                                <DashboardIcon />
                            </ListItemIcon>
                            {open && <ListItemText primary="Dashboard" />}
                        </ListItemButton>
                    </ListItem>
                    <Divider />

                    {/* C) “Flex grow” spacer to push the next items to the bottom */}
                    <Box sx={{ flexGrow: 1 }} />

                    <Divider />

                    {/* D) Bottom items */}
                    <ListItem disablePadding>
                        <ListItemButton component={Link} to="/settings">
                            <ListItemIcon sx={{ color: theme.palette.common.white }}>
                                <SettingsIcon />
                            </ListItemIcon>
                            {open && <ListItemText primary="Settings" />}
                        </ListItemButton>
                    </ListItem>

                    <ListItem disablePadding>
                        <ListItemButton component={Link} to="/help">
                            <ListItemIcon sx={{ color: theme.palette.common.white }}>
                                <HelpOutlineIcon />
                            </ListItemIcon>
                            {open && <ListItemText primary="Help" />}
                        </ListItemButton>
                    </ListItem>
                </List>
            </Box>
        </Drawer>
    );
};

export default ResponsiveSidebar;

===== ..\src\pages\dashboardPage.jsx =====
import React, {useState} from 'react';
import {Responsive, WidthProvider} from 'react-grid-layout';
import {widgetRegistry} from '../widgets/WidgetRegistry';
import 'react-grid-layout/css/styles.css';
import 'react-resizable/css/styles.css';
import AddWidgetButton from '../components/dashboard/addWidgetButton';
import IconButton from '@mui/material/IconButton';
import DeleteIcon from '@mui/icons-material/Delete';
import WidgetPreviewPopup from "../components/dashboard/WidgetPreviewPopup";

const ResponsiveGridLayout = WidthProvider(Responsive);

function renderWidget(widget) {
    const widgetMeta = widgetRegistry[widget.type];
    if (!widgetMeta) {
        return <div>Unknown widget type: {widget.type}</div>;
    }
    const WidgetComponent = widgetMeta.component;
    return <WidgetComponent config={widget.config} />;
}

function DashboardPage() {
    const [widgets, setWidgets] = useState([]);
    const [layout, setLayout] = useState([]);
    const [previewOpen, setPreviewOpen] = useState(false);

    const addWidget = (widgetType) => {
        const newId = 'widget-' + Date.now();
        const widgetMeta = widgetRegistry[widgetType];

        setWidgets((prev) => [
            ...prev,
            { id: newId, type: widgetType, config: {} }
        ]);

        setLayout((prev) => [
            ...prev,
            {
                i: newId,
                x: 0,
                y: 0,
                w: widgetMeta.minW  || 4,
                h: widgetMeta.minH || 4,
                minW: widgetMeta.minW || 2,
                minH: widgetMeta.minH || 2
            }
        ]);
    };

    const removeWidget = (id) => {
        setWidgets((prev) => {
            return prev.filter((w) => w.id !== id);
        });

        setLayout((prev) => {
            return prev.filter((item) => item.i !== id);
        });
    };

    const openWidgetSelectionDialog = () => {
        setPreviewOpen(true);
    };

    const handleWidgetSelect = (widgetType) => {
        addWidget(widgetType);
        setPreviewOpen(false);
    };

    return (
        <div style={{ margin: '1rem' , width: '100%'}}>
            <h2>Dashboard</h2>

            {/* The grid layout */}
            <ResponsiveGridLayout
                draggableCancel=".MuiIconButton-root, .no-drag"
                className="layout"
                layouts={{ lg: layout, md: layout, sm: layout, xs: layout, xxs: layout }}
                breakpoints={{ lg: 1200, md: 996, sm: 768, xs: 480, xxs: 0 }}
                cols={{ lg: 12, md: 12, sm: 12, xs: 12, xxs: 12 }}
                rowHeight={35}
                onLayoutChange={(currentLayout) => {
                    setLayout(currentLayout);
                }}
            >
                {widgets.map((widget) => {
                    // Find the layout object for this widget by matching the id.
                    const layoutItem = layout.find((item) => item.i === widget.id);
                    return (
                        <div
                            key={widget.id}
                            style={{
                                border: '1px solid #444',
                                background: '#1e1e1e',
                                position: 'relative',
                            }}
                        >
                            <IconButton
                                onClick={() => removeWidget(widget.id)}
                                sx={{
                                    position: 'absolute',
                                    top: 0,
                                    right: 0,
                                    zIndex: 5,
                                    cursor: 'pointer',
                                    color: '#fff',
                                }}
                            >
                                <DeleteIcon />
                            </IconButton>
                            {renderWidget(widget)}
                            {/* Debug overlay showing the current width (w) and height (h) */}
                            {layoutItem && (
                                <div
                                    style={{
                                        position: 'absolute',
                                        bottom: '2px',
                                        left: '2px',
                                        fontSize: '10px',
                                        color: 'white',
                                        background: 'rgba(0, 0, 0, 0.5)',
                                        padding: '2px 4px',
                                        borderRadius: '2px',
                                    }}
                                >
                                    {`w: ${layoutItem.w}, h: ${layoutItem.h}`}
                                </div>
                            )}
                        </div>
                    );
                })}
            </ResponsiveGridLayout>

            <AddWidgetButton onClick={openWidgetSelectionDialog} />

            <WidgetPreviewPopup
                open={previewOpen}
                onClose={() => setPreviewOpen(false)}
                onSelect={handleWidgetSelect}
            />
        </div>
    );
}

export default DashboardPage;

import React, { useState, useEffect } from 'react';
import {

const generateInitialData = () => {
    const data = [];
    const now = new Date();
    // Generate one data point per minute for the past 24 hours (1440 minutes)
    for (let i = 1440; i >= 0; i--) {
        const time = new Date(now.getTime() - i * 60000);
        data.push({
            time: time.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),

import React, { useState, useEffect } from 'react';
import {
    ResponsiveContainer,
    LineChart,
    Line,
    XAxis,
    YAxis,
    CartesianGrid,
    Tooltip,
    Legend,
    Brush
} from 'recharts';
import { Card, CardContent, Typography } from '@mui/material';

// Mapping numeric values to operation mode labels.
const modeMapping = {
    0: "Off",
    1: "Error State",
    2: "Setup Mode",
    3: "Auto (Not Started)",
    4: "Auto (Started & Idle)",
    5: "Auto (In Production)"
};

const generateInitialData = () => {
    const data = [];
    const now = new Date();
    // Generate one data point per minute for the past 2 hours (120 minutes) as an example.
    const numPoints = 120;
    for (let i = numPoints; i >= 0; i--) {
        const time = new Date(now.getTime() - i * 60000);
        // For demonstration, randomly pick a mode from 0 to 5.
        const mode = Math.floor(Math.random() * 6);
        data.push({
            time: time.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
            mode
        });
    }
    return data;
};

const OperationModeWidget = ({ config }) => {
    const [data, setData] = useState(generateInitialData());

    // Update every minute by shifting out the oldest value and adding a new one.
    useEffect(() => {
        const interval = setInterval(() => {
            setData(prevData => {
                const newData = [...prevData];
                newData.shift();
                const now = new Date();
                const mode = Math.floor(Math.random() * 6);
                newData.push({
                    time: now.toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
                    mode
                });
                return newData;
            });
        }, 60000);
        return () => clearInterval(interval);
    }, []);

    // Tooltip formatter to show text labels.
    const tooltipFormatter = (value, name) => {
        if (name === "mode") {
            return modeMapping[value];
        }
        return value;
    };

    // Y-axis tick formatter to display mode labels.
    const yTickFormatter = (value) => modeMapping[value];

    // Generate ticks for all possible modes.
    const yTicks = [0, 1, 2, 3, 4, 5];

    return (
        <Card style={{ width: '100%', height: '100%' }}>
            <CardContent style={{ display: 'flex', flexDirection: 'column', height: '100%' }}>
                <Typography variant="h6" gutterBottom>
                    Operation Mode Over Time
                </Typography>
                <div style={{ flexGrow: 1 }}>
                    <ResponsiveContainer width="100%" height="100%">
                        <LineChart data={data}>
                            <CartesianGrid strokeDasharray="3 3" />
                            <XAxis dataKey="time" minTickGap={20} />
                            <YAxis tickFormatter={yTickFormatter} ticks={yTicks} domain={[0, 5]} />
                            <Tooltip formatter={tooltipFormatter} />
                            <Legend />
                            <Line
                                type="step"
                                dataKey="mode"
                                stroke="#82ca9d"
                                strokeWidth={2}
                                name="Operation Mode"
                                dot
                            />
                            <Brush className="no-drag" dataKey="time" height={30} stroke="#82ca9d" />

                    </ResponsiveContainer>

    );
